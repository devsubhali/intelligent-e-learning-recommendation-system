    {
      "cell_type": "code",
      "source": [
        "def recommend_courses(user_input, top_n=5):\n",
        "    user_vector = vectorizer.transform([user_input])\n",
        "\n",
        "    if user_vector.nnz == 0:\n",
        "        print(\"\\nSorry, no relevant courses were found for that interest.\")\n",
        "        print(\"Try another topic such as Python, Design, Business, or Machine Learning.\")\n",
        "        return None\n",
        "\n",
        "    user_similarity = cosine_similarity(user_vector, tfidf_matrix)[0]\n",
        "\n",
        "    top_indices = np.argsort(user_similarity)[::-1][:top_n]\n",
        "\n",
        "    recommendations = df.iloc[top_indices].copy()\n",
        "\n",
        "    recommendations[\"similarity_score\"] = (\n",
        "        user_similarity[top_indices] * 100\n",
        "    ).round(2)\n",
